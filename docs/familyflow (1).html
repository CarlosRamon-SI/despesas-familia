Instalar dependências npm install whatsapp-web.js qrcode-terminal // 2. familyflow-bot.js const { Client, LocalAuth } = require('whatsapp-web.js'); const qrcode = require('qrcode-terminal'); const client = new Client({ authStrategy: new LocalAuth() }); client.on('qr', qr => qrcode.generate(qr, {small:true})); client.on('ready', () => console.log('Bot conectado!')); client.on('message', async msg => { const [cmd, val, ...rest] = msg.body.trim().split(' '); const desc = rest.join(' '); if (cmd === 'despesa') { await api.post('/transactions', { type:'expense', value:parseFloat(val), desc }); msg.reply(`✅ Despesa R$${val} — ${desc}`); } if (cmd === 'receita') { await api.post('/transactions', { type:'income', value:parseFloat(val), desc }); msg.reply(`✅ Receita R$${val} — ${desc}`); } if (cmd === '/saldo') { const { balance } = await api.get('/balance'); msg.reply(`💰 Saldo: R$ ${balance}`); } if (cmd === '/resumo') { const r = await api.get('/summary'); msg.reply(`📊 Receitas: R$${r.income}\n❌ Despesas: R$${r.expense}`); } }); client.initialize();
